import React, { useState } from 'react';
import { CheckCircle, Upload, Ruler, Users, MapPin, Phone, Mail, ChevronRight, ChevronLeft } from 'lucide-react';
import { motion } from 'framer-motion';

const App = () => {
  const [currentStep, setCurrentStep] = useState(1);
  const [measurements, setMeasurements] = useState({
    chest: '',
    waist: '',
    hips: '',
    shoulder: '',
    sleeveLength: '',
    inseam: '',
    outseam: '',
    neck: ''
  });
  const [designImage, setDesignImage] = useState(null);
  const [selectedFabric, setSelectedFabric] = useState(null);
  const [customerDetails, setCustomerDetails] = useState({
    name: '',
    address: '',
    whatsapp: '',
    email: ''
  });
  const [isSubmitted, setIsSubmitted] = useState(false);

  const fabrics = [
    { id: 1, name: 'Premium Wool', color: 'bg-gray-800', price: '₹2500/meter' },
    { id: 2, name: 'Italian Silk', color: 'bg-amber-700', price: '₹3200/meter' },
    { id: 3, name: 'Cotton Blend', color: 'bg-blue-600', price: '₹1800/meter' },
    { id: 4, name: 'Linen', color: 'bg-stone-400', price: '₹2200/meter' },
    { id: 5, name: 'Velvet', color: 'bg-purple-800', price: '₹2800/meter' },
    { id: 6, name: 'Tweed', color: 'bg-green-700', price: '₹2400/meter' },
    { id: 7, name: 'Cashmere', color: 'bg-yellow-600', price: '₹4500/meter' },
    { id: 8, name: 'Twill', color: 'bg-indigo-800', price: '₹2000/meter' }
  ];

  const handleMeasurementChange = (field, value) => {
    setMeasurements(prev => ({
      ...prev,
      [field]: value
    }));
  };

  const handleFileUpload = (e) => {
    const file = e.target.files[0];
    if (file) {
      setDesignImage(URL.createObjectURL(file));
    }
  };

  const handleCustomerDetailChange = (field, value) => {
    setCustomerDetails(prev => ({
      ...prev,
      [field]: value
    }));
  };

  const handleSubmit = () => {
    // In a real application, this would send data to your backend
    // For now, we'll just show the success page
    setIsSubmitted(true);
    
    // Simulate sending data to WhatsApp
    const orderData = {
      measurements,
      fabric: selectedFabric,
      customer: customerDetails,
      designImage: designImage ? 'Uploaded' : 'Not provided'
    };
    
    console.log('Order Data:', orderData);
  };

  const MeasurementForm = () => (
    <motion.div
      initial={{ opacity: 0, y: 20 }}
      animate={{ opacity: 1, y: 0 }}
      className="space-y-6"
    >
      <div className="bg-white rounded-xl shadow-lg p-6">
        <h2 className="text-2xl font-bold text-gray-800 mb-6 flex items-center">
          <Ruler className="mr-2 text-blue-600" />
          Body Measurements
        </h2>
        <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
          {Object.entries(measurements).map(([key, value]) => (
            <div key={key} className="space-y-2">
              <label className="block text-sm font-medium text-gray-700 capitalize">
                {key.replace(/([A-Z])/g, ' $1')}
              </label>
              <input
                type="number"
                value={value}
                onChange={(e) => handleMeasurementChange(key, e.target.value)}
                className="w-full px-4 py-3 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 focus:border-transparent transition-all"
                placeholder={`Enter ${key.replace(/([A-Z])/g, ' $1')} measurement`}
              />
            </div>
          ))}
        </div>
      </div>
    </motion.div>
  );

  const DesignUpload = () => (
    <motion.div
      initial={{ opacity: 0, y: 20 }}
      animate={{ opacity: 1, y: 0 }}
      className="space-y-6"
    >
      <div className="bg-white rounded-xl shadow-lg p-6">
        <h2 className="text-2xl font-bold text-gray-800 mb-6 flex items-center">
          <Upload className="mr-2 text-blue-600" />
          Suit Design Reference
        </h2>
        <div className="border-2 border-dashed border-gray-300 rounded-lg p-8 text-center hover:border-blue-400 transition-colors">
          <input
            type="file"
            accept="image/*"
            onChange={handleFileUpload}
            className="hidden"
            id="design-upload"
          />
          <label htmlFor="design-upload" className="cursor-pointer">
            <Upload className="mx-auto h-12 w-12 text-gray-400 mb-4" />
            <p className="text-lg font-medium text-gray-700 mb-2">Upload Design Reference</p>
            <p className="text-sm text-gray-500">PNG, JPG, or PDF up to 10MB</p>
          </label>
        </div>
        {designImage && (
          <div className="mt-4">
            <img src={designImage} alt="Design preview" className="max-w-full h-48 object-contain rounded-lg" />
          </div>
        )}
      </div>
    </motion.div>
  );

  const FabricSelection = () => (
    <motion.div
      initial={{ opacity: 0, y: 20 }}
      animate={{ opacity: 1, y: 0 }}
      className="space-y-6"
    >
      <div className="bg-white rounded-xl shadow-lg p-6">
        <h2 className="text-2xl font-bold text-gray-800 mb-6">Select Fabric</h2>
        <div className="flex space-x-4 overflow-x-auto pb-4">
          {fabrics.map((fabric) => (
            <div
              key={fabric.id}
              onClick={() => setSelectedFabric(fabric)}
              className={`flex-shrink-0 w-48 cursor-pointer rounded-lg border-2 p-4 transition-all ${
                selectedFabric?.id === fabric.id
                  ? 'border-blue-500 bg-blue-50'
                  : 'border-gray-200 hover:border-gray-300'
              }`}
            >
              <div className={`${fabric.color} w-full h-20 rounded-lg mb-3`}></div>
              <h3 className="font-semibold text-gray-800">{fabric.name}</h3>
              <p className="text-sm text-gray-600">{fabric.price}</p>
            </div>
          ))}
        </div>
        {selectedFabric && (
          <div className="mt-4 p-4 bg-green-50 rounded-lg border border-green-200">
            <p className="text-green-800 font-medium">Selected: {selectedFabric.name}</p>
          </div>
        )}
      </div>
    </motion.div>
  );

  const CustomerDetails = () => (
    <motion.div
      initial={{ opacity: 0, y: 20 }}
      animate={{ opacity: 1, y: 0 }}
      className="space-y-6"
    >
      <div className="bg-white rounded-xl shadow-lg p-6">
        <h2 className="text-2xl font-bold text-gray-800 mb-6 flex items-center">
          <Users className="mr-2 text-blue-600" />
          Your Details
        </h2>
        <div className="space-y-4">
          <div>
            <label className="block text-sm font-medium text-gray-700 mb-2">Full Name</label>
            <input
              type="text"
              value={customerDetails.name}
              onChange={(e) => handleCustomerDetailChange('name', e.target.value)}
              className="w-full px-4 py-3 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 focus:border-transparent transition-all"
              placeholder="Enter your full name"
            />
          </div>
          <div>
            <label className="block text-sm font-medium text-gray-700 mb-2">Address</label>
            <textarea
              value={customerDetails.address}
              onChange={(e) => handleCustomerDetailChange('address', e.target.value)}
              className="w-full px-4 py-3 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 focus:border-transparent transition-all"
              placeholder="Enter your complete address"
              rows="3"
            />
          </div>
          <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
            <div>
              <label className="block text-sm font-medium text-gray-700 mb-2">WhatsApp Number</label>
              <div className="relative">
                <Phone className="absolute left-3 top-3.5 h-4 w-4 text-gray-400" />
                <input
                  type="tel"
                  value={customerDetails.whatsapp}
                  onChange={(e) => handleCustomerDetailChange('whatsapp', e.target.value)}
                  className="w-full pl-10 px-4 py-3 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 focus:border-transparent transition-all"
                  placeholder="Enter WhatsApp number"
                />
              </div>
            </div>
            <div>
              <label className="block text-sm font-medium text-gray-700 mb-2">Email Address</label>
              <div className="relative">
                <Mail className="absolute left-3 top-3.5 h-4 w-4 text-gray-400" />
                <input
                  type="email"
                  value={customerDetails.email}
                  onChange={(e) => handleCustomerDetailChange('email', e.target.value)}
                  className="w-full pl-10 px-4 py-3 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 focus:border-transparent transition-all"
                  placeholder="Enter your email"
                />
              </div>
            </div>
          </div>
        </div>
      </div>
    </motion.div>
  );

  const SuccessPage = () => (
    <motion.div
      initial={{ opacity: 0, scale: 0.8 }}
      animate={{ opacity: 1, scale: 1 }}
      className="min-h-screen bg-gradient-to-br from-blue-50 to-indigo-100 flex items-center justify-center p-4"
    >
      <div className="bg-white rounded-2xl shadow-2xl p-8 text-center max-w-md w-full">
        <motion.div
          initial={{ scale: 0 }}
          animate={{ scale: 1 }}
          transition={{ delay: 0.2, type: "spring", stiffness: 200 }}
          className="w-20 h-20 bg-green-100 rounded-full flex items-center justify-center mx-auto mb-6"
        >
          <CheckCircle className="w-12 h-12 text-green-600" />
        </motion.div>
        <h1 className="text-3xl font-bold text-gray-800 mb-4">Order Registered!</h1>
        <p className="text-gray-600 mb-6">
          Your custom suit order has been successfully registered. Our team will contact you shortly via WhatsApp for payment and final confirmation.
        </p>
        <div className="bg-blue-50 rounded-lg p-4 mb-6">
          <p className="text-sm text-blue-800">
            <strong>Order Details Sent To:</strong><br />
            {customerDetails.whatsapp}
          </p>
        </div>
        <button
          onClick={() => {
            setIsSubmitted(false);
            setCurrentStep(1);
            setMeasurements({
              chest: '',
              waist: '',
              hips: '',
              shoulder: '',
              sleeveLength: '',
              inseam: '',
              outseam: '',
              neck: ''
            });
            setDesignImage(null);
            setSelectedFabric(null);
            setCustomerDetails({
              name: '',
              address: '',
              whatsapp: '',
              email: ''
            });
          }}
          className="w-full bg-blue-600 text-white py-3 rounded-lg font-medium hover:bg-blue-700 transition-colors"
        >
          Place Another Order
        </button>
      </div>
    </motion.div>
  );

  if (isSubmitted) {
    return <SuccessPage />;
  }

  return (
    <div className="min-h-screen bg-gradient-to-br from-gray-50 to-blue-50">
      {/* Header */}
      <div className="bg-white shadow-sm">
        <div className="max-w-6xl mx-auto px-4 py-6">
          <h1 className="text-3xl font-bold text-gray-800">Custom Suit Tailoring</h1>
          <p className="text-gray-600 mt-2">Perfect fit, perfect style - tailored just for you</p>
        </div>
      </div>

      {/* Tutorial Video Section */}
      <div className="max-w-6xl mx-auto px-4 py-8">
        <div className="bg-white rounded-xl shadow-lg overflow-hidden">
          <div className="bg-gradient-to-r from-blue-600 to-indigo-700 p-6">
            <h2 className="text-2xl font-bold text-white flex items-center">
              <Ruler className="mr-2" />
              How to Take Your Measurements
            </h2>
            <p className="text-blue-100 mt-2">Watch this video to learn how to measure yourself accurately</p>
          </div>
          <div className="p-6">
            <div className="bg-gray-900 aspect-video rounded-lg flex items-center justify-center">
              <div className="text-center text-white">
                <div className="w-16 h-16 bg-red-600 rounded-full flex items-center justify-center mx-auto mb-4">
                  <div className="w-0 h-0 border-l-8 border-l-transparent border-r-8 border-r-transparent border-b-8 border-b-white ml-1"></div>
                </div>
                <p className="text-lg font-medium">Measurement Tutorial Video</p>
                <p className="text-sm text-gray-300 mt-2">Click to play</p>
              </div>
            </div>
          </div>
        </div>
      </div>

      {/* Progress Steps */}
      <div className="max-w-6xl mx-auto px-4 mb-8">
        <div className="flex items-center justify-between">
          {[1, 2, 3, 4].map((step) => (
            <div key={step} className="flex items-center">
              <div
                className={`w-10 h-10 rounded-full flex items-center justify-center font-semibold ${
                  currentStep >= step
                    ? 'bg-blue-600 text-white'
                    : 'bg-gray-200 text-gray-600'
                }`}
              >
                {step}
              </div>
              <div className="ml-2 text-sm font-medium text-gray-700">
                {step === 1 && 'Measurements'}
                {step === 2 && 'Design'}
                {step === 3 && 'Fabric'}
                {step === 4 && 'Details'}
              </div>
              {step < 4 && (
                <ChevronRight className="mx-4 text-gray-400" />
              )}
            </div>
          ))}
        </div>
      </div>

      {/* Main Content */}
      <div className="max-w-6xl mx-auto px-4 pb-8">
        {currentStep === 1 && <MeasurementForm />}
        {currentStep === 2 && <DesignUpload />}
        {currentStep === 3 && <FabricSelection />}
        {currentStep === 4 && <CustomerDetails />}

        {/* Navigation Buttons */}
        <div className="flex justify-between mt-8">
          <button
            onClick={() => setCurrentStep(Math.max(1, currentStep - 1))}
            disabled={currentStep === 1}
            className={`flex items-center px-6 py-3 rounded-lg font-medium transition-colors ${
              currentStep === 1
                ? 'bg-gray-200 text-gray-500 cursor-not-allowed'
                : 'bg-gray-600 text-white hover:bg-gray-700'
            }`}
          >
            <ChevronLeft className="mr-2" />
            Previous
          </button>
          
          {currentStep < 4 ? (
            <button
              onClick={() => setCurrentStep(currentStep + 1)}
              className="flex items-center px-6 py-3 bg-blue-600 text-white rounded-lg font-medium hover:bg-blue-700 transition-colors"
            >
              Next
              <ChevronRight className="ml-2" />
            </button>
          ) : (
            <button
              onClick={handleSubmit}
              disabled={
                !customerDetails.name ||
                !customerDetails.address ||
                !customerDetails.whatsapp
              }
              className={`px-8 py-3 rounded-lg font-medium transition-colors ${
                !customerDetails.name ||
                !customerDetails.address ||
                !customerDetails.whatsapp
                  ? 'bg-gray-300 text-gray-500 cursor-not-allowed'
                  : 'bg-green-600 text-white hover:bg-green-700'
              }`}
            >
              Submit Order
            </button>
          )}
        </div>
      </div>

      {/* Footer */}
      <footer className="bg-gray-800 text-white py-8 mt-12">
        <div className="max-w-6xl mx-auto px-4 text-center">
          <h3 className="text-xl font-bold mb-2">Premium Tailoring Services</h3>
          <p className="text-gray-300">Crafting bespoke suits with precision and style</p>
          <div className="flex justify-center space-x-6 mt-4">
            <div className="flex items-center">
              <Phone className="mr-2" />
              <span>+91 98765 43210</span>
            </div>
            <div className="flex items-center">
              <Mail className="mr-2" />
              <span>orders@customtailor.com</span>
            </div>
          </div>
        </div>
      </footer>
    </div>
  );
};

export default App;
